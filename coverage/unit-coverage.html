import (
        "fmt"
        "os"

        "github.com/elsyahtech/gorest"
)

// AppConfig loads application settings from environment variables and validates the environment.
func AppConfig() (gorest.Config, error) {
        cfg := gorest.ConfigDefault

        var err error

        cfg.AppName = str("GOREST_APP_NAME", gorest.DefaultAppName)
        cfg.Version = str("GOREST_APP_VERSION", "v1.0.0")
        cfg.Environment = str("GOREST_APP_ENV", gorest.Development)

        if err := validateEnvironment(cfg.Environment); err != nil {
                return gorest.Config{}, fmt.Errorf("%w", err)
        }

        cfg.Timezone = str("GOREST_APP_TIMEZONE", gorest.DefaultTimezone)

        cfg.Debug, err = strToBool("GOREST_APP_DEBUG", false)
        if err != nil {
                return gorest.Config{}, fmt.Errorf("failed to parse GOREST_APP_DEBUG: %w", err)
        }

        if value, ok := os.LookupEnv("GOREST_APP_CHECK_FOR_UPDATES"); ok && value != "" {
                checkForUpdates, err := strToBool("GOREST_APP_CHECK_FOR_UPDATES", true)
                if err != nil {
                        return gorest.Config{}, fmt.Errorf("failed to parse GOREST_APP_CHECK_FOR_UPDATES: %w", err)
                }

                cfg.CheckForUpdates = &checkForUpdates
        }

        return cfg, nil
}

// DatabaseConfig loads the MySQL connection settings and optional startup tasks.
func DatabaseConfig() (database.Config, error) {
        cfg := database.ConfigDefault

        var err error

        cfg.Disable, err = strToBool("GOREST_DB_DISABLE", database.DatabaseDisabled)
        if err != nil {
                return database.Config{}, fmt.Errorf("failed to parse GOREST_DB_DISABLE: %w", err)
        }

        // This starter supports MySQL only.
        cfg.Driver = str("GOREST_DB_DRIVER", database.MYSQL)
        if !strings.EqualFold(cfg.Driver, database.MYSQL) {
                return database.Config{}, fmt.Errorf("GOREST_DB_DRIVER must be %q for the MySQL starter", database.MYSQL)
        }

        cfg.Driver = database.MYSQL
        cfg.Host = str("GOREST_DB_HOST", database.DefaultDatabaseHost)
        cfg.Port = str("GOREST_DB_PORT", database.MYSQL_PORT)
        cfg.Name = str("GOREST_DB_NAME", database.DefaultDatabaseName)

        cfg.Username = str("GOREST_DB_USER", "")
        if cfg.Username == "" {
                return database.Config{}, errors.New("DB Username cannot be empty")
        }

        cfg.Password = str("GOREST_DB_PASSWORD", "")
        if cfg.Password == "" {
                return database.Config{}, errors.New("DB Password cannot be empty")
        }

        cfg.AllowInsecureProduction, err = strToBool("GOREST_DB_ALLOW_INSECURE_PRODUCTION", false)
        if err != nil {
                return database.Config{}, fmt.Errorf("failed to parse GOREST_DB_ALLOW_INSECURE_PRODUCTION: %w", err)
        }

        cfg.TLS = str("GOREST_DB_TLS", database.TLSDisabled)
        cfg.SSLCertPath = str("GOREST_DB_CERT_PATH", "")

        cfg.SSLCAPath = str("GOREST_DB_CA_PATH", "")
        // MySQL TLS requires a CA path in this starter.
        if cfg.Driver == database.MYSQL && cfg.TLS == database.TLSTrue && cfg.SSLCAPath == "" {
                return database.Config{}, errors.New("GOREST_DB_CA_PATH cannot be empty when MySQL TLS is enabled")
        }

        cfg.SSLKeyPath = str("GOREST_DB_KEY_PATH", "")
        cfg.Charset = str("GOREST_DB_CHARSET", database.DefaultCharset)
        cfg.Timeout = strToTimeDuration("GOREST_DB_TIMEOUT", database.DefaultTimeout)
        cfg.ConnectionMaxLifetime = strToTimeDuration("GOREST_DB_CONNECTION_MAX_LIFETIME", database.DefaultConnectionMaxLifetime)
        cfg.MaxOpenConnections = strToInt("GOREST_DB_MAX_OPEN_CONNECTIONS", database.DefaultMaxOpenConnections)
        cfg.MaxIdleConnections = strToInt("GOREST_DB_MAX_IDLE_CONNECTIONS", database.DefaultMaxIdleConnections)

        cfg.QueryLogging, err = strToBool("GOREST_DB_QUERY_LOGGING", database.DefaultQueryLogging)
        if err != nil {
                return database.Config{}, fmt.Errorf("failed to parse GOREST_DB_QUERY_LOGGING: %w", err)
        }

        cfg.DatabaseDirectory = str("GOREST_DB_DIRECTORY", database.DefaultDatabaseDirectory)

        cfg.Migration, err = strToBool("GOREST_DB_MIGRATION", database.MigrationDisabled)
        if err != nil {
                return database.Config{}, fmt.Errorf("failed to parse GOREST_DB_MIGRATION: %w", err)
        }

        cfg.Seeder, err = strToBool("GOREST_DB_SEEDER", database.SeederDisabled)
        if err != nil {
                return database.Config{}, fmt.Errorf("failed to parse GOREST_DB_SEEDER: %w", err)
        }

        cfg.MigrationDirectory = str("GOREST_DB_MIGRATIONS_DIRECTORY", database.DefaultMigrationDirectory)
        cfg.SeederDirectory = str("GOREST_DB_SEEDERS_DIRECTORY", database.DefaultSeederDirectory)

        return cfg, nil
}

import (

func envString(key string) string {
        return strings.TrimSpace(os.Getenv(key))
}

func envDuration(key string) (time.Duration, error) {
        value := envString(key)
        if value == "" {
                return 0, nil
        }

        duration, err := time.ParseDuration(value)
        if err != nil {
                return 0, fmt.Errorf("%s must be a valid Go duration: %w", key, err)
        }

        if duration < 0 {
                return 0, fmt.Errorf("%s cannot be negative", key)
        }

        return duration, nil
}

func envNonNegativeInt(key string) (int, error) {
        value := envString(key)
        if value == "" {
                return 0, nil
        }

        number, err := strconv.Atoi(value)
        if err != nil || number < 0 {
                return 0, fmt.Errorf("%s must be a non-negative integer", key)
        }

        return number, nil
}

func envFileMode(key string) (os.FileMode, error) {
        value := envString(key)
        if value == "" {
                return 0, nil
        }

        mode, err := strconv.ParseUint(value, 8, 32)
        if err != nil || mode > 0o777 {
                return 0, fmt.Errorf("%s must be an octal permission mode between 0000 and 0777", key)
        }

        return os.FileMode(mode), nil
}

func envTLSMinVersion(key string) (uint16, error) {
        switch strings.ToLower(envString(key)) {
        case "":
                return 0, nil
        case "1.2", "tls1.2", "tlsv1.2":
                return tls.VersionTLS12, nil
        case "1.3", "tls1.3", "tlsv1.3":
                return tls.VersionTLS13, nil
        default:
                return 0, fmt.Errorf("%s must be 1.2 or 1.3", key)
        }
}

func str(key, defaultValue string) string {
        value := envString(key)
        if value == "" {
                return defaultValue
        }

        return value
}

func strPtr(key, defaultValue string) *string {
        value := str(key, defaultValue)

        return &value
}

func strToInt(key string, defaultValue int) int {
        value := envString(key)
        if value == "" {
                return defaultValue
        }

        parsed, err := strconv.Atoi(value)
        if err != nil {
                return defaultValue
        }

        return parsed
}

func strToTimeDuration(key string, defaultValue time.Duration) time.Duration {
        value := envString(key)
        if value == "" {
                return defaultValue
        }

        parsed, err := time.ParseDuration(value)
        if err != nil {
                return defaultValue
        }

        return parsed
}

func toStrArray(key string, defaultValue []string) []string {
        value, ok := os.LookupEnv(key)
        if !ok {
                return defaultValue
        }

        value = strings.TrimSpace(value)

        if value == "" {
                return make([]string, 0)
        }

        items := strings.Split(value, ",")

        result := make([]string, 0, len(items))

        for _, item := range items {
                if trimmed := strings.TrimSpace(item); trimmed != "" {
                        result = append(result, trimmed)
                }
        }

        return result
}

func strToBool(key string, defaultValue bool) (bool, error) {
        value := strings.TrimSpace(os.Getenv(key))
        if value == "" {
                return defaultValue, nil
        }

        parsed, err := strconv.ParseBool(value)
        if err != nil {
                return false, fmt.Errorf("%s must be a boolean: %w", key, err)
        }

        return parsed, nil
}

func validateEnvironment(value string) error {
        switch value {
        case gorest.Development, gorest.Production, gorest.Testing:
                return nil
        default:
                return fmt.Errorf("GOREST_APP_ENV must be %s, %s, or %s", gorest.Development, gorest.Production, gorest.Testing)
        }
}

// RedisConfig loads optional Redis settings; Redis is disabled by default.
func RedisConfig() (redis.Config, error) {
        cfg := redis.ConfigDefault

        var err error

        cfg.Disable, err = strToBool("GOREST_REDIS_DISABLE", true)
        if err != nil {
                return redis.Config{}, fmt.Errorf("failed to parse GOREST_REDIS_DISABLE: %w", err)
        }

        cfg.Database = strToInt("GOREST_REDIS_DATABASE", redis.DefaultRedisDatabase)
        cfg.Host = str("GOREST_REDIS_HOST", redis.DefaultRedisHost)
        cfg.Port = str("GOREST_REDIS_PORT", redis.DefaultRedisPort)
        cfg.Username = str("GOREST_REDIS_USER", "")
        cfg.Password = str("GOREST_REDIS_PASSWORD", "")

        cfg.TLS, err = strToBool("GOREST_REDIS_TLS", false)

                return redis.Config{}, fmt.Errorf("failed to parse GOREST_REDIS_TLS: %w", err)
        }

        cfg.SSLCertPath = str("GOREST_REDIS_CERT_PATH", "")
        if cfg.SSLCertPath == "" {
                if cfg.TLS {
                        // This starter requires a certificate path when Redis TLS is enabled.
                        return redis.Config{}, errors.New("GOREST_REDIS_CERT_PATH cannot be empty when GOREST_REDIS_TLS is true")
                }

        cfg.SSLKeyPath = str("GOREST_REDIS_KEY_PATH", "")
        cfg.SSLCAPath = str("GOREST_REDIS_CA_PATH", "")
        cfg.Timeout = strToTimeDuration("GOREST_REDIS_TIMEOUT", redis.DefaultRedisTimeout)
        cfg.PoolTimeout = strToTimeDuration("GOREST_REDIS_POOL_TIMEOUT", redis.DefaultRedisPoolTimeout)
        cfg.ConnMaxLifetime = strToTimeDuration("GOREST_REDIS_CONN_MAX_LIFETIME", redis.DefaultRedisConnMaxLifetime)
        cfg.ConnMaxIdleTime = strToTimeDuration("GOREST_REDIS_CONN_MAX_IDLE_TIME", redis.DefaultRedisConnMaxIdleTime)

        cfg.AllowInsecureProduction, err = strToBool("GOREST_REDIS_ALLOW_INSECURE_PRODUCTION", false)
        if err != nil {
                return redis.Config{}, fmt.Errorf("failed to parse GOREST_REDIS_ALLOW_INSECURE_PRODUCTION: %w", err)
        }

        cfg.MaxRetries = strToInt("GOREST_REDIS_MAX_RETRIES", redis.DefaultRedisMaxRetries)
        cfg.PoolSize = strToInt("GOREST_REDIS_POOL_SIZE", redis.DefaultRedisPoolSize)
        cfg.MinIdleConns = strToInt("GOREST_REDIS_MIN_IDLE_CONNS", redis.DefaultRedisMinIdleConns)
        cfg.MaxActiveConns = strToInt("GOREST_REDIS_MAX_ACTIVE_CONNS", redis.DefaultRedisMaxActiveConns)
        cfg.ReadBufferSize = strToInt("GOREST_REDIS_READ_BUFFER_SIZE", redis.DefaultRedisReadBufferSize)
        cfg.WriteBufferSize = strToInt("GOREST_REDIS_WRITE_BUFFER_SIZE", redis.DefaultRedisWriteBufferSize)

        return cfg, nil
}

// ServerConfig loads HTTP server, middleware, and listener settings.
// Authentication and authorization must be configured separately for protected routes.
func ServerConfig() (server.Config, error) {
        cfg := server.ConfigDefault

        cfg.Host = str("GOREST_SERVER_HOST", server.DefaultServerHost)
        cfg.Port = strToInt("GOREST_SERVER_PORT", server.DefaultServerPort)

        stackTrace, err := strToBool("GOREST_SERVER_RECOVER_ENABLE_STACK_TRACE", false)
        if err != nil {
                return server.Config{}, fmt.Errorf("failed to parse GOREST_SERVER_RECOVER_ENABLE_STACK_TRACE: %w", err)
        }

        cfg.Recover = &recover.Config{
                EnableStackTrace: stackTrace,
        }

        cfg.Requestid = &requestid.Config{
                Header: os.Getenv("GOREST_SERVER_REQUEST_ID_HEADER"),
        }

        cfg.Compress = &compress.Config{
                Level: compress.Level(strToInt("GOREST_SERVER_COMPRESS_LEVEL", int(compress.LevelDefault))),
        }

        cfg.Core, err = fiberCoreConfig()
        if err != nil {
                return server.Config{}, fmt.Errorf("failed to configure fiber core: %w", err)
        }

        origins := toStrArray("GOREST_SERVER_CORS_ORIGINS", []string{"http://localhost:3000", "http://127.0.0.1:3000"})
        if len(origins) == 0 {
                origins = []string{"http://localhost:3000", "http://127.0.0.1:3000"}
        }

        methods := toStrArray("GOREST_SERVER_CORS_METHODS", []string{"GET", "HEAD", "POST", "PUT", "PATCH", "DELETE", "OPTIONS"})
        if len(methods) == 0 {
                methods = []string{"GET", "HEAD", "POST", "PUT", "PATCH", "DELETE", "OPTIONS"}
        }

        headers := toStrArray("GOREST_SERVER_CORS_HEADERS", []string{"Origin", "Content-Type", "Accept", "Authorization", "X-Request-ID"})
        if len(headers) == 0 {
                headers = []string{"Origin", "Content-Type", "Accept", "Authorization", "X-Request-ID"}
        }

        allowCredentials, err := strToBool("GOREST_SERVER_CORS_ALLOW_CREDENTIALS", false)
        if err != nil {
                return server.Config{}, fmt.Errorf("failed to parse GOREST_SERVER_CORS_ALLOW_CREDENTIALS: %w", err)
        }

        allowPrivateNetwork, err := strToBool("GOREST_SERVER_CORS_ALLOW_PRIVATE_NETWORK", false)
        if err != nil {
                return server.Config{}, fmt.Errorf("failed to parse GOREST_SERVER_CORS_ALLOW_PRIVATE_NETWORK: %w", err)
        }

        disableCORSValueRedaction, err := strToBool("GOREST_SERVER_CORS_DISABLE_VALUE_REDACTION", false)
        if err != nil {
                return server.Config{}, fmt.Errorf("failed to parse GOREST_SERVER_CORS_DISABLE_VALUE_REDACTION: %w", err)
        }

        cfg.Cors = &cors.Config{
                AllowOrigins:          origins,
                AllowMethods:          methods,
                AllowHeaders:          headers,
                ExposeHeaders:         toStrArray("GOREST_SERVER_CORS_EXPOSE_HEADERS", make([]string, 0)),
                MaxAge:                strToInt("GOREST_SERVER_CORS_MAX_AGE", 300),
                AllowCredentials:      allowCredentials,
                AllowPrivateNetwork:   allowPrivateNetwork,
                DisableValueRedaction: disableCORSValueRedaction,
        }

        hSTSExcludeSubdomains, err := strToBool("GOREST_SERVER_HELMET_HSTS_EXCLUDE_SUBDOMAINS", false)
        if err != nil {
                return server.Config{}, err
        }

        cspReportOnly, err := strToBool("GOREST_SERVER_HELMET_CSP_REPORT_ONLY", false)
        if err != nil {
                return server.Config{}, err
        }

        hSTSPreloadEnabled, err := strToBool("GOREST_SERVER_HELMET_HSTS_PRELOAD_ENABLED", false)
        if err != nil {
                return server.Config{}, err
        }

        cfg.Helmet = &helmet.Config{
                XSSProtection:             os.Getenv("GOREST_SERVER_HELMET_XSS_PROTECTION"),
                ContentTypeNosniff:        os.Getenv("GOREST_SERVER_HELMET_CONTENT_TYPE_NOSNIFF"),
                XFrameOptions:             os.Getenv("GOREST_SERVER_HELMET_X_FRAME_OPTIONS"),
                ContentSecurityPolicy:     os.Getenv("GOREST_SERVER_HELMET_CONTENT_SECURITY_POLICY"),
                ReferrerPolicy:            os.Getenv("GOREST_SERVER_HELMET_REFERRER_POLICY"),
                PermissionPolicy:          os.Getenv("GOREST_SERVER_HELMET_PERMISSION_POLICY"),
                CrossOriginEmbedderPolicy: os.Getenv("GOREST_SERVER_HELMET_CROSS_ORIGIN_EMBEDDER_POLICY"),
                CrossOriginOpenerPolicy:   os.Getenv("GOREST_SERVER_HELMET_CROSS_ORIGIN_OPENER_POLICY"),
                CrossOriginResourcePolicy: os.Getenv("GOREST_SERVER_HELMET_CROSS_ORIGIN_RESOURCE_POLICY"),
                OriginAgentCluster:        os.Getenv("GOREST_SERVER_HELMET_ORIGIN_AGENT_CLUSTER"),
                XDNSPrefetchControl:       os.Getenv("GOREST_SERVER_HELMET_X_DNS_PREFETCH_CONTROL"),
                XDownloadOptions:          os.Getenv("GOREST_SERVER_HELMET_X_DOWNLOAD_OPTIONS"),
                XPermittedCrossDomain:     os.Getenv("GOREST_SERVER_HELMET_X_PERMITTED_CROSS_DOMAIN"),
                HSTSMaxAge:                strToInt("GOREST_SERVER_HELMET_HSTS_MAX_AGE", 0),
                HSTSExcludeSubdomains:     hSTSExcludeSubdomains,
                CSPReportOnly:             cspReportOnly,
                HSTSPreloadEnabled:        hSTSPreloadEnabled,
        }

        skipFailedRequests, err := strToBool("GOREST_SERVER_RATE_LIMIT_SKIP_FAILED_REQUESTS", false)
        if err != nil {
                return server.Config{}, err
        }

        skipSuccessfulRequests, err := strToBool("GOREST_SERVER_RATE_LIMIT_SKIP_SUCCESSFUL_REQUESTS", false)
        if err != nil {
                return server.Config{}, err
        }

        disableHeaders, err := strToBool("GOREST_SERVER_RATE_LIMIT_DISABLE_HEADERS", false)
        if err != nil {
                return server.Config{}, err
        }

        disableValueRedaction, err := strToBool("GOREST_SERVER_RATE_LIMIT_DISABLE_VALUE_REDACTION", false)
        if err != nil {
                return server.Config{}, err
        }

        cfg.Limiter = &limiter.Config{
                Max:                    strToInt("GOREST_SERVER_RATE_LIMIT_MAX", 50),
                Expiration:             strToTimeDuration("GOREST_SERVER_RATE_LIMIT_EXPIRATION", time.Minute),
                SkipFailedRequests:     skipFailedRequests,
                SkipSuccessfulRequests: skipSuccessfulRequests,
                DisableHeaders:         disableHeaders,
                DisableValueRedaction:  disableValueRedaction,
        }

        listener, err := loadServerListenerConfig()
        if err != nil {
                return server.Config{}, err
        }

        cfg.Listener = listener

        return cfg, nil
}

func fiberCoreConfig() (*fiber.Config, error) {
        strictRouting, err := strToBool("GOREST_SERVER_STRICT_ROUTING", false)
        if err != nil {
                return &fiber.Config{}, fmt.Errorf("failed to parse GOREST_SERVER_STRICT_ROUTING: %w", err)
        }

        caseSensitive, err := strToBool("GOREST_SERVER_CASE_SENSITIVE", false)
        if err != nil {
                return &fiber.Config{}, fmt.Errorf("failed to parse GOREST_SERVER_CASE_SENSITIVE: %w", err)
        }

        skipUnmatchedRoutes, err := strToBool("GOREST_SERVER_SKIP_UNMATCHED_ROUTES", false)
        if err != nil {
                return &fiber.Config{}, fmt.Errorf("failed to parse GOREST_SERVER_SKIP_UNMATCHED_ROUTES: %w", err)
        }

        disableHeadAutoRegister, err := strToBool("GOREST_SERVER_DISABLE_HEAD_AUTO_REGISTER", false)
        if err != nil {
                return &fiber.Config{}, fmt.Errorf("failed to parse GOREST_SERVER_DISABLE_HEAD_AUTO_REGISTER: %w", err)
        }

        disableFlashMessages, err := strToBool("GOREST_SERVER_DISABLE_FLASH_MESSAGES", false)
        if err != nil {
                return &fiber.Config{}, fmt.Errorf("failed to parse GOREST_SERVER_DISABLE_FLASH_MESSAGES: %w", err)
        }

        immutable, err := strToBool("GOREST_SERVER_IMMUTABLE", false)
        if err != nil {
                return &fiber.Config{}, fmt.Errorf("failed to parse GOREST_SERVER_IMMUTABLE: %w", err)
        }

        unescapePath, err := strToBool("GOREST_SERVER_UNESCAPE_PATH", false)
        if err != nil {
                return &fiber.Config{}, fmt.Errorf("failed to parse GOREST_SERVER_UNESCAPE_PATH: %w", err)
        }

        passLocalsToViews, err := strToBool("GOREST_SERVER_PASS_LOCALS_TO_VIEWS", false)
        if err != nil {
                return &fiber.Config{}, fmt.Errorf("failed to parse GOREST_SERVER_PASS_LOCALS_TO_VIEWS: %w", err)
        }

        passLocalsToContext, err := strToBool("GOREST_SERVER_PASS_LOCALS_TO_CONTEXT", false)
        if err != nil {
                return &fiber.Config{}, fmt.Errorf("failed to parse GOREST_SERVER_PASS_LOCALS_TO_CONTEXT: %w", err)
        }

        getOnly, err := strToBool("GOREST_SERVER_GET_ONLY", false)
        if err != nil {
                return &fiber.Config{}, fmt.Errorf("failed to parse GOREST_SERVER_GET_ONLY: %w", err)
        }

func loadServerListenerConfig() (server.ListenerConfig, error) {
        listener := server.ListenerConfig{
                Network:           envString("GOREST_SERVER_LISTENER_NETWORK"),
                TLSCertFile:       envString("GOREST_SERVER_CERT_FILE"),
                TLSCertKeyFile:    envString("GOREST_SERVER_CERT_KEY_FILE"),
                TLSCertClientFile: envString("GOREST_SERVER_CERT_CLIENT_FILE"),
                UnixSocketPath:    envString("GOREST_SERVER_UNIX_SOCKET_PATH"),
        }

        var err error

        if listener.PreforkRecoverInterval, err = envDuration("GOREST_SERVER_PREFORK_RECOVER_INTERVAL"); err != nil {
                return server.ListenerConfig{}, err
        }

        if listener.ShutdownTimeout, err = envDuration("GOREST_SERVER_SHUTDOWN_TIMEOUT"); err != nil {
                return server.ListenerConfig{}, err
        }

        if listener.PreforkShutdownGracePeriod, err = envDuration("GOREST_SERVER_PREFORK_SHUTDOWN_GRACE_PERIOD"); err != nil {
                return server.ListenerConfig{}, err
        }

        if listener.PreforkRecoverThreshold, err = envNonNegativeInt("GOREST_SERVER_PREFORK_RECOVER_THRESHOLD"); err != nil {
                return server.ListenerConfig{}, err
        }

        if listener.UnixSocketFileMode, err = envFileMode("GOREST_SERVER_UNIX_SOCKET_FILE_MODE"); err != nil {
                return server.ListenerConfig{}, err
        }

        if listener.TLSMinVersion, err = envTLSMinVersion("GOREST_SERVER_TLS_MIN_VERSION"); err != nil {
                return server.ListenerConfig{}, err
        }

        if listener.EnablePrefork, err = strToBool("GOREST_SERVER_ENABLE_PREFORK", false); err != nil {
                return server.ListenerConfig{}, err
        }

        if listener.EnablePrintRoutes, err = strToBool("GOREST_SERVER_ENABLE_PRINT_ROUTES", false); err != nil {
                return server.ListenerConfig{}, err
        }

        return listener, nil
}

package library

import (
        "errors"
        "fmt"
        "net/http"

        "github.com/google/uuid"

        "github.com/elsyahtech/gorest"
        "github.com/elsyahtech/gorest-api-starter/app/entity"
        "github.com/elsyahtech/gorest/helper"
)

func (lib *UserLibrary) Login(ctx *gorest.Context, req *entity.User) (string, int, error) {
        if req == nil {
                return "User payload must not be empty", http.StatusBadRequest,
                        errors.New("add user: request is empty")
        }

        // Get user by email from the database.
        user, message, httpCode, err := lib.UserModel.GetUserByEmail(req.Email)
        if err != nil {
                return message, httpCode, fmt.Errorf("get user by email: %w", err)
        }

        // Validate the password using Argon2 hashing algorithm.
        passwordMatch, message, httpCode, err := helper.VerifyArgon(req.Password, user.Password)
        if err != nil {
                return message, httpCode, fmt.Errorf("verify password: %w", err)
        }

        if !passwordMatch {
                return "", http.StatusUnauthorized, errors.New("ensure you have entered the correct email and password")
        }

        // Generate a unique token ID for the user using uuid.
        // TokenID also can be used from user.ID.
        // Can be adjustable based on your application needs.
        tokenID := uuid.New().String() // Generate a unique token ID for the user.

        // Generate token authentication and store it in cookies.
        message, httpCode, err = lib.App.Filter().CreateToken(*ctx, tokenID)
        if err != nil {
                return message, httpCode, fmt.Errorf("create token: %w", err)
        }

        return message, httpCode, nil
}

func (lib *UserLibrary) CreateUser(req *entity.User) (string, int, error) {
        if req == nil {
                return "User payload must not be empty", http.StatusBadRequest,
                        errors.New("add user: request is empty")
        }

        passwordHashed, message, httpCode, err := helper.GenerateArgon(req.Password)
        if err != nil {
                return message, httpCode, fmt.Errorf("%w", err)
        }

        req.Password = passwordHashed

        message, httpCode, err = lib.UserModel.CreateUser(req)
        if err != nil {
                return message, httpCode, fmt.Errorf("%w", err)
        }

        return message, httpCode, nil
}

func (lib *UserLibrary) GetUsers() (*[]entity.User, string, int, error) {
        result, message, httpCode, err := lib.UserModel.GetUsers()
        if err != nil {
                return nil, message, httpCode, fmt.Errorf("%w", err)
        }

        return result, message, httpCode, nil
}

package model

import (
        "errors"
        "fmt"
        "net/http"
        "time"

        "github.com/elsyahtech/gorest"
        "github.com/elsyahtech/gorest-api-starter/app/entity"
)

func (model *UserModel) CreateUser(req *entity.User) (string, int, error) {
        if req == nil {
                return "User payload must not be empty", http.StatusBadRequest,
                        errors.New("add user: payload is empty")
        }

        database := model.App.Database()

        transaction, message, httpCode, err := database.BeginTx()
        if err != nil {
                return message, httpCode, fmt.Errorf("%w", err)
        }

        defer func() {
                if message, _, err := transaction.Rollback(); err != nil {
                        model.App.Log(gorest.Map{"error": err}).Error(message)
                }
        }()

        timeNow := time.Now()

        req.CreatedAt = timeNow

        query := transaction.Table("users").Create(req)
        if query.Error != nil {
                return query.Message, query.HTTPCode, fmt.Errorf("%w", query.Error)
        }

        if message, httpCode, err = transaction.Commit(); err != nil {
                return message, httpCode, fmt.Errorf("commit transaction failed: %w", err)
        }

        return query.Message, query.HTTPCode, nil
}

func (model *UserModel) GetUserByEmail(email string) (*entity.User, string, int, error) {
        if email == "" {
                return nil, "Email param must not be empty", http.StatusBadRequest,
                        errors.New("get user by email: email is empty")
        }

        database := model.App.Database()

        var user entity.User

        query := database.Table("users").
                Select("*").
                Where("users.email = ? AND users.is_active = ?", email, true).
                Find(&user)
        if query.Error != nil {
                return nil, query.Message, query.HTTPCode, fmt.Errorf("%w", query.Error)
        }

        return &user, query.Message, query.HTTPCode, nil
}

func (model *UserModel) GetUsers() (*[]entity.User, string, int, error) {
        database := model.App.Database()

        var users []entity.User

        query := database.Table("users").Find(&users)
        if query.Error != nil {
                return nil, query.Message, query.HTTPCode, fmt.Errorf("%w", query.Error)
        }

        return &users, query.Message, query.HTTPCode, nil
}

        group.Post("/login", hndlr.Login)

// BuildUserResponse converts a user into its JSON response DTO.
func BuildUserResponse(response *entity.User) *entity.UserResponse {
        if response == nil {
                return nil
        }

        return &entity.UserResponse{
                ID:        response.ID,
                Email:     response.Email,
                IsActive:  response.IsActive,
                CreatedAt: helper.ConvertTimeToString(response.CreatedAt),
                UpdatedAt: helper.ConvertTimePointerToStringPointer(response.UpdatedAt),
                DeletedAt: helper.ConvertTimePointerToStringPointer(response.DeletedAt),
        }
}

// BuildUserResponseArray converts users into response DTOs.
// It returns an empty array when responses is nil or empty.
func BuildUserResponseArray(responses *[]entity.User) *[]entity.UserResponse {
        if responses == nil || len(*responses) == 0 {
                users := make([]entity.UserResponse, 0)

                return &users
        }

        res := make([]entity.UserResponse, 0, len(*responses))

        for _, item := range *responses {
                res = append(res, *BuildUserResponse(&item))
        }

        return &res
}
